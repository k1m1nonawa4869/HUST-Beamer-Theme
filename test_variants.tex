        \raggedright For every $z \in \mathcal{X}$, $\innerp{x - \mathrm{Proj}_{\mathcal{X}}(x)}{\mathrm{Proj}_{\mathcal{X}}(x) - z} \ge 0$.
      \end{lemma}

\subsection{Custom Lists (enumitem \& tasks)}
\begin{frame}[t]{Custom Lists with enumitem and tasks}{HUST square numbered badges and multi-column horizontal lists}
  \small
  \begin{definition}[Pareto Optimality]
    \begin{hustsqlist}
      \item $x^* \in C$ is \emph{Pareto optimal} if $\nexists x \in C$ such that $g_i(x) \le g_i(x^*)$ for all $i$ and $g_j(x) < g_j(x^*)$ for at least one index $j$. \label{def:pareto}
      \item $\bar{x} \in C$ is \emph{weakly Pareto optimal} if $\nexists x \in C$ such that $g_i(x) < g_i(\bar{x})$ for all $i$.
      \item The set of all Pareto optimal decisions is the \emph{Pareto set}, and its image is the \emph{Pareto front}.
    \end{hustsqlist}
  \end{definition}

      \begin{algorithm}[H]
        \scriptsize
        \caption{MPG-E Algorithm}\label{algo:mpg}
        \KwIn{$x_0 \in \mathcal{X}$, $\beta \in (0,1)$, $\gamma > 0$}
        \KwOut{Weak Pareto solution $x^*$}
        \BlankLine
        \For{$k = 0, 1, 2, \dots$}{
          $d_k = \argmin_{d}\!\big\{\max_i \innerp{\nabla f_i(x_k)}{d} + \tfrac{1}{2}\norm{d}^2\big\}$\;
          \If{$\norm{d_k} < \epsilon$}{
            \Return $x_k$\;
          }
          $x_{k+1} = \mathrm{Proj}_{\mathcal{X}}(x_k + \alpha_k d_k)$\;
        }
      \end{algorithm}
      \end{minipage}
    \end{column}

\subsection{Multi-Objective Formulation}
\begin{frame}{Multi-Objective Optimization Study}{End-to-end cross-referencing and automated footnote citations}
  \vspace{-12pt}
  Consider the constrained multi-objective problem over $C \subset \mathbb{R}^n$:
  \begin{equation}
    \min_{x \in C} \; F(x) = \big(f_1(x), f_2(x), \dots, f_m(x)\big) \label{eq:mop}
  \end{equation}
  Bello-Cruz et al.\cite{bellocruz2025} solve \Cref{eq:mop} under the \cref{def:pareto} optimality criterion:
  \begin{itemize}[nosep]
    \item \textbf{Model \& geometry}: Problem \Cref{eq:mop} connects to the analytic contour in \cref{fig:cauchy} via Cauchy's formula \eqref{eq:cauchy}.
    \item \textbf{Algorithm \& code}: The inertial scheme\cite{alvarez2024} is formalized in \cref{algo:mpg} and implemented in \cref{lst:python} and \cref{lst:cpp}.
    \item \textbf{Empirical evaluation}: Benchmarks in \cref{tab:exp} confirm Pareto convergence\cite{hust2026} matching \cref{def:pareto}.
    \item \textbf{Smart footer}: Citation strip clears the footer logo (\texttt{footerlogo=hust}) and truncates long entries cleanly with \texttt{...}.
  \end{itemize}
\end{frame}
