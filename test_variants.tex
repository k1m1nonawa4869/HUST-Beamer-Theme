\subsection{Môi trường định nghĩa \& khối ghi chú}
\begin{frame}{Môi trường định nghĩa và khối ghi chú}{Đồng bộ màu sắc nhận diện và chú thích tự động}
  \begin{definition}[Phép toán hai ngôi]
    Cho tập $X \neq \varnothing$. Một \note{phép toán hai ngôi}{Một ánh xạ từ $X^2 \to X$ gắn mỗi cặp phần tử $(x, y)$ thành một phần tử $x * y \in X$.} trên tập $X$, ký hiệu là $*$, là một quy tắc biến đổi mỗi cặp phần tử $(x, y) \in X^2$ thành phần tử $x * y \in X$. Nói cách khác, phép toán $*$ là một ánh xạ:
    \[
      (*) : X^2 \longrightarrow X, \quad (x, y) \longmapsto x * y
    \]
  \end{definition}

  \begin{exampleblock}{Nhận xét quan trọng}
    \warning Các môi trường khối và toán học chuẩn của Beamer (\texttt{block}, \texttt{definition}, \texttt{theorem}, \texttt{exampleblock}) được tinh chỉnh tự động theo bảng màu thương hiệu HUST, hỗ trợ đánh số và tham chiếu chéo thông minh qua \texttt{cleveref}.
  \end{exampleblock}
\end{frame}

\subsection{Tùy biến danh sách (enumitem \& tasks)}
\begin{frame}[t]{Tùy biến danh sách với enumitem và tasks}{Đánh số dạng ô vuông nhận diện thương hiệu HUST}
  \small
  \begin{definition}[Nghiệm Pareto]
    \begin{hustsqlist}
      \item $x^* \in C$ là \emph{nghiệm Pareto} nếu $\nexists x \in C$ sao cho $g_i(x) \le g_i(x^*)$ với mọi $i$ và $g_j(x) < g_j(x^*)$ tại ít nhất một chỉ số $j$. \label{def:pareto}
      \item $\bar{x} \in C$ là \emph{nghiệm Pareto yếu} nếu $\nexists x \in C$ sao cho $g_i(x) < g_i(\bar{x})$ với mọi $i$.
      \item Tập hợp tất cả các nghiệm Pareto được gọi là \emph{tập Pareto}. Ảnh của nó trong không gian mục tiêu được gọi là \emph{biên Pareto}.
    \end{hustsqlist}
  \end{definition}
  \vskip 2pt

      \begin{algorithm}[H]
        \scriptsize
        \caption{Thuật toán MPG-E}\label{algo:mpg}
        \KwIn{$x_0 \in \mathcal{X}$, $\beta \in (0,1)$, $\gamma > 0$}
        \KwOut{Nghiệm Pareto yếu $x^*$}
        \BlankLine
        \For{$k = 0, 1, 2, \dots$}{
          $d_k = \argmin_{d} \left\{ \max_i \innerp{\nabla f_i(x_k)}{d} + \frac{1}{2}\norm{d}^2 \right\}$\;
          \If{$\norm{d_k} < \epsilon$}{
            \Return $x_k$\;
          }
          $x_{k+1} = \mathrm{Proj}_{\mathcal{X}}(x_k + \alpha_k d_k)$\;
        }
      \end{algorithm}
      \end{minipage}
    \end{column}

def mpg_step(x, grad, alpha=0.01):
    d_k = -np.mean(grad(x), axis=0)
    return x + alpha * d_k

\subsection{Phương pháp tối ưu hóa đa mục tiêu}
\begin{frame}{Nghiên cứu tối ưu hóa đa mục tiêu}{Hệ thống tham chiếu chéo toàn diện và trích dẫn tự động}
  \vspace{-12pt}
  Xét bài toán tối ưu hóa đa mục tiêu với tập ràng buộc $C \subset \mathbb{R}^n$:
  \begin{equation}
    \min_{x \in C} \; F(x) = \big(f_1(x), f_2(x), \dots, f_m(x)\big) \label{eq:mop}
  \end{equation}
  Nghiên cứu của Bello-Cruz và cộng sự\cite{bellocruz2025} giải quyết \Cref{eq:mop} theo tiêu chuẩn \cref{def:pareto}:
  \begin{itemize}[nosep]
    \item \textbf{Mô hình \& hình học}: Bài toán \Cref{eq:mop} liên kết với miền giải tích \cref{fig:cauchy} qua công thức tích phân vi phân \eqref{eq:cauchy}.
    \item \textbf{Thuật toán \& cài đặt}: Quy trình giải lặp quán tính\cite{alvarez2024} được đặc tả tại \cref{algo:mpg}, mã nguồn mẫu tại \cref{lst:python} và \cref{lst:cpp}.
    \item \textbf{Hiệu năng thực nghiệm}: Bảng so sánh \cref{tab:exp} khẳng định thuật toán hội tụ về nghiệm Pareto \cref{def:pareto} trên hệ thống chuẩn\cite{hust2026}.
    \item \textbf{Chân trang thông minh}: Cụm trích dẫn tránh logo (\texttt{footerlogo=hust}), tự động rút gọn \texttt{...} trên 1 dòng.
  \end{itemize}
\end{frame}

  \small
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{1. Phần mềm \& Ứng dụng hỗ trợ}:
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{TeXstudio} (Khuyến nghị sử dụng offline).
        \item \textbf{MiKTeX / TeX Live} (Hệ phân phối LaTeX).
        \item \textbf{Overleaf}: Hoạt động trực tiếp trên trình duyệt.
      \end{itemize}
      \vskip 0.3em
      \textbf{2. Trình biên dịch bắt buộc}:
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{LuaLaTeX} (Khuyến nghị số 1) hoặc \textbf{XeLaTeX} để nạp phông hệ thống và tiếng Việt Unicode.
        \item Không dùng \texttt{pdfLaTeX} (không nạp được OTF/TTF).
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{3. Trình xử lý tài liệu tham khảo}:
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Sử dụng \textbf{Biber} kết hợp gói \texttt{biblatex}.
        \item Quy trình biên dịch:
          \[
            \text{Lua\LaTeX} \longrightarrow \text{Biber} \longrightarrow \text{Lua\LaTeX}
          \]
      \end{itemize}
      \vskip 0.3em
      \textbf{4. Cấu hình TeXstudio (1 chạm)}:
      \begin{itemize}\scriptsize\setlength{\itemsep}{2pt}
        \item \textit{Options $\to$ Configure TeXstudio $\to$ Build}:
        \item \textbf{Default Compiler}: chọn \texttt{LuaLaTeX} (thêm cờ \texttt{-shell-escape} và cài \textbf{Python} nếu dùng \texttt{minted}).
        \item \textbf{Default Bibliography}: chọn \texttt{Biber}.
        \item Nhấn \textbf{F5} để biên dịch toàn bộ.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
